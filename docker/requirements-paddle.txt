# PaddleOCR install for Jetson — pinned and carefully curated.
#
# We pin to the 2.x line because:
#   - paddlepaddle 3.x segfaults inside its PIR loader on aarch64 Jetson
#   - paddleocr 2.10+ pulls albumentations / python-docx which would clobber
#     the GStreamer-enabled system cv2 (installed via apt python3-opencv)
#
# Everything here is installed with --no-deps in the Dockerfile so that
# pip doesn't decide to "helpfully" install opencv-contrib-python or
# upgrade numpy past 2.x.

paddlepaddle==2.6.2
paddleocr==2.7.3

# transitive deps that paddleocr 2.7 actually needs at import time

astor>=0.8
# paddlepaddle 2.6.2 imports opt_einsum unconditionally in paddle.tensor.einsum.
# It pins opt-einsum==3.3.0 in its own metadata; using a different version

# at runtime on some einsum codepaths.
